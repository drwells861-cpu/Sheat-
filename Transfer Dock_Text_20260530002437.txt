        function extractProductNames(data) {
            if (!data || data.length < 2) {
                showToast('الملف فارغ أو لا يحتوي على بيانات كافية.', 'error');
                return;
            }

            const headers = data[0].map(h => String(h).trim().toLowerCase());
            let productColIndex = headers.findIndex(h => h.includes('اسم المنتج') || h.includes('product') || h.includes('الاسم') || h.includes('name'));
            
            if (productColIndex === -1) {
                productColIndex = 0; 
                showToast('لم نجد عمود "اسم المنتج"، تم اعتماد العمود الأول.', 'warning');
            }

            const productNames = [];
            for (let i = 1; i < data.length; i++) {
                if (data[i] && data[i][productColIndex]) {
                    const name = String(data[i][productColIndex]).trim();
                    if (name) productNames.push(name);
                }
            }

            if (productNames.length === 0) {
                showToast('لم يتم العثور على أي أسماء منتجات في الملف.', 'error');
                return;
            }

            startProcessing(productNames);
        }

        async function fetchWithRetry(payload, retries = 3) {
            const currentKey = getApiKey();
            
            // القائمة الشاملة لجميع نماذج جوجل الممكنة (النظام سيختبرها تلقائياً)
            const modelsToTry = successfulModel ? [successfulModel] : [
                'gemini-2.5-flash-preview-09-2025',
                'gemini-2.5-flash',
                'gemini-2.0-flash',
                'gemini-1.5-flash-latest',
                'gemini-1.5-flash-8b',
                'gemini-1.5-flash',
                'gemini-1.5-pro'
            ];
            
            const delays = [1000, 3000, 6000];
            let lastErrorMessage = "فشل الاتصال بجميع النماذج";

            for (let i = 0; i < retries; i++) {
                for (const model of modelsToTry) {
                    const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${currentKey}`;
                    try {
                        const response = await fetch(endpoint, {
                            method: 'POST',
                            headers: { 'Content-Type': 'application/json' },
                            body: JSON.stringify(payload)
                        });

                        if (!response.ok) {
                            let errorDetails = "";
                            try {
                                const errJson = await response.json();
                                errorDetails = errJson.error?.message || "خطأ غير معروف";
                            } catch(e) {
                                errorDetails = `HTTP ${response.status}`;
                            }
                            
                            // أخطاء التوثيق (المفتاح خطأ) نتوقف فوراً
                            if (response.status === 400 && errorDetails.toLowerCase().includes('api key')) {
                                 throw new Error(`مفتاح API غير صالح`);
                            }
                            if (response.status === 403) {
                                throw new Error(`المفتاح لا يملك صلاحيات`);
                            }
                            
                            // إذا كان النموذج غير مدعوم، نتجاهل الخطأ ونجرب النموذج الذي يليه
                            if (response.status === 404 || errorDetails.includes('not found') || errorDetails.includes('is not supported')) {
                                lastErrorMessage = `النموذج ${model} غير مدعوم بحسابك`;
                                continue; 
                            }
                            
                            throw new Error(errorDetails);
                        }

                        const result = await response.json();
                        const textContent = result.candidates?.[0]?.content?.parts?.[0]?.text;
                        if (!textContent) throw new Error("استجابة فارغة من الذكاء الاصطناعي");
                        
                        // حفظ النموذج الناجح لتسريع العمليات القادمة وعدم التجربة من الصفر
                        successfulModel = model;
                        
                        try {
                            return JSON.parse(textContent);
                        } catch (parseError) {
                            const cleaned = textContent.replace(/```json/g, '').replace(/```/g, '').trim();
                            return JSON.parse(cleaned);
                        }

                    } catch (err) {
                        lastErrorMessage = err.message;
                        // إذا كان الخطأ متعلقاً بصلاحية المفتاح نفسه، نوقف كل شيء فوراً
                        if (err.message.includes('مفتاح API') || err.message.includes('صلاحيات')) {
                            throw err; 
                        }
                        // عدا ذلك، نستمر لتجربة النموذج التالي
                    }
                }
                
                // إذا فشلت كل النماذج في هذه المحاولة (بسبب ضغط الشبكة مثلا)، ننتظر ثم نعيد المحاولة
                if (i < retries - 1) {
                    await new Promise(res => setTimeout(res, delays[i]));
                }
            }
            throw new Error(lastErrorMessage);
        }

            // Prompt تم تحسينه وإضافة شروط صارمة للتصنيف الدقيق
            const prompt = `أنت صيدلي وخبير بيانات منتجات. قم بتحليل هذا المنتج بدقة: "${productName}".
يجب أن ترجع النتيجة كـ JSON.
شروط هامة:
1. التصنيف الرئيسي (mainCategory): اختر تصنيفاً عاماً دقيقاً (أمثلة: أدوية، مكملات غذائية، مستحضرات تجميل، مستلزمات طبية، عناية شخصية).
2. التصنيف الفرعي (subCategory): اختر تصنيفاً فرعياً دقيقاً طبياً أو تجارياً (أمثلة: مضاد حيوي، مسكن ألم، خافض حرارة، فيتامين، غسول وجه، قطرة عين).
3. رابط الصورة (imageUrl): اكتب دائماً القيمة "لا يوجد".`;

            const payload = {
                contents: [{ parts: [{ text: prompt }] }],

                    temperature: 0.1,

                        type: "OBJECT",
                        properties: {

                    <td class="px-6 py-4">
                        <span class="inline-block bg-indigo-50 text-indigo-700 border border-indigo-100 rounded-md px-2 py-1 text-[10px] font-bold whitespace-normal">
                            ${enrichedItem['المادة الفعالة']}
                        </span>
                    </td>
                    <td class="px-6 py-4 text-slate-700 font-medium text-[11px]">

            } catch (error) {
                // الكشف الذكي عن نوع الخطأ وعرضه بوضوح
                let errorDisplay = error.message;
                
                extractedDataList.push({
                    'اسم المنتج': productName, 'الوصف': errorDisplay, 'المادة الفعالة': '-', 'التصنيف الرئيسي': '-', 'التصنيف الفرعي': '-', 'رابط الصورة': ''
                });

                tr.innerHTML = `
